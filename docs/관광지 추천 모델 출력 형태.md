\## 모델과 장소임베딩 로드



```

import os

import pandas as pd

import numpy as np

from catboost import CatBoostRegressor

from fastapi import FastAPI, HTTPException

from pydantic import BaseModel, Field

from typing import List, Optional



app = FastAPI(title="STOG 전라북도 여행지 맞춤 추천 API", version="2.0")



\# ---------------------------------------------------------

\# 1. 서버 스타트업 시 모델 \& 장소 임베딩 로드

\# ---------------------------------------------------------

BASE\_DIR = os.path.expanduser('\~/travel\_data')

MODEL\_PATH = os.path.join(BASE\_DIR, 'STOG\_model\_merged.cbm')

EMB\_PATH = os.path.join(BASE\_DIR, 'jeonbuk\_place\_embeddings\_merged.csv')



print("🚀 STOG AI 추천 모델 및 임베딩 메모리 로딩 중...")

model = CatBoostRegressor()

model.load\_model(MODEL\_PATH)

place\_df = pd.read\_csv(EMB\_PATH)

print(f"✅ 로딩 완료: 총 {len(place\_df):,}개 전북 장소 임베딩 준비 완료!")



\# ---------------------------------------------------------

\# 2. API 입력 데이터 규격 (직접 6축 성향 입력받기)

\# ---------------------------------------------------------

class RecommendRequest(BaseModel):

&#x20;   # 유저 6축 성향 직접 입력 (0.0 \~ 1.0 실수)

&#x20;   user\_nature: float = Field(0.5, ge=0.0, le=1.0, description="자연 선호도")

&#x20;   user\_culture: float = Field(0.5, ge=0.0, le=1.0, description="역사문화 선호도")

&#x20;   user\_food: float = Field(0.5, ge=0.0, le=1.0, description="미식 선호도")

&#x20;   user\_shopping: float = Field(0.5, ge=0.0, le=1.0, description="쇼핑 선호도")

&#x20;   user\_experience: float = Field(0.5, ge=0.0, le=1.0, description="체험 선호도")

&#x20;   user\_relaxation: float = Field(0.5, ge=0.0, le=1.0, description="휴식 선호도")

&#x20;   

&#x20;   # 인구통계 및 상황 특성 (선택사항, CatBoost 모델 피처용)

&#x20;   gender: str = Field("여", description="성별 ('남', '여')")

&#x20;   age\_grp: int = Field(30, description="연령대 (20, 30, 40...)")

&#x20;   income: int = Field(4, description="소득 구간 (1\~8)")

&#x20;   travel\_mission\_priority: int = Field(1, description="여행 1순위 미션 번호")

&#x20;   travel\_motive\_1: int = Field(1, description="여행 동기 번호 (1\~10)")

&#x20;   travel\_companions\_num: int = Field(2, description="동행 인원 수")

&#x20;   

&#x20;   # 필터 및 추천 개수

&#x20;   gungu\_filter: Optional\[str] = Field(None, description="시/군 필터 (예: '전주시', 없으면 전북 전체)")

&#x20;   top\_n: int = Field(5, ge=1, le=20, description="추천받을 장소 수")



\# ---------------------------------------------------------

\# 3. API 엔드포인트 구현 (/api/recommend)

\# ---------------------------------------------------------

@app.post("/api/recommend")

def recommend\_places(req: RecommendRequest):

&#x20;   # 1) 입력받은 유저 6축 벡터 배열화

&#x20;   u\_vec\_dict = {

&#x20;       'nature': req.user\_nature,

&#x20;       'culture': req.user\_culture,

&#x20;       'food': req.user\_food,

&#x20;       'shopping': req.user\_shopping,

&#x20;       'experience': req.user\_experience,

&#x20;       'relaxation': req.user\_relaxation

&#x20;   }

&#x20;   

&#x20;   u\_vec = np.array(\[u\_vec\_dict\['nature'], u\_vec\_dict\['culture'], u\_vec\_dict\['food'], 

&#x20;                     u\_vec\_dict\['shopping'], u\_vec\_dict\['experience'], u\_vec\_dict\['relaxation']])

&#x20;   

&#x20;   df = place\_df.copy()

&#x20;   

&#x20;   # 지역(시/군) 선택 필터링

&#x20;   if req.gungu\_filter:

&#x20;       df = df\[df\['장소명'].str.contains(req.gungu\_filter, na=False)].copy()

&#x20;       if len(df) == 0:

&#x20;           df = place\_df.copy() # 필터 검색 결과 없으면 전체 대상으로 복원

&#x20;           

&#x20;   # 2) 코사인 유사도 (match\_score) 연산

&#x20;   p\_vecs = df\[\['자연\_임베딩', '역사문화\_임베딩', '미식\_임베딩', '쇼핑\_임베딩', '체험\_임베딩', '휴식\_임베딩']].values

&#x20;   dot\_prods = np.sum(u\_vec \* p\_vecs, axis=1)

&#x20;   norms = np.linalg.norm(u\_vec) \* np.linalg.norm(p\_vecs, axis=1)

&#x20;   df\['match\_score'] = np.where(norms > 0, dot\_prods / norms, 0.5)

&#x20;   

&#x20;   # 3) CatBoost 피처 생성

&#x20;   df\['p\_nature'] = df\['자연\_임베딩']

&#x20;   df\['p\_culture'] = df\['역사문화\_임베딩']

&#x20;   df\['p\_food'] = df\['미식\_임베딩']

&#x20;   df\['p\_shopping'] = df\['쇼핑\_임베딩']

&#x20;   df\['p\_experience'] = df\['체험\_임베딩']

&#x20;   df\['p\_relaxation'] = df\['휴식\_임베딩']

&#x20;   

&#x20;   # 유저 성향 피처 주입

&#x20;   df\['nature'] = u\_vec\_dict\['nature']

&#x20;   df\['culture'] = u\_vec\_dict\['culture']

&#x20;   df\['food'] = u\_vec\_dict\['food']

&#x20;   df\['shopping'] = u\_vec\_dict\['shopping']

&#x20;   df\['experience'] = u\_vec\_dict\['experience']

&#x20;   df\['relaxation'] = u\_vec\_dict\['relaxation']

&#x20;   

&#x20;   # 인구통계 상황 피처 주입

&#x20;   df\['AGE\_GRP'] = req.age\_grp

&#x20;   df\['INCOME'] = req.income

&#x20;   df\['TRAVEL\_MISSION\_PRIORITY'] = req.travel\_mission\_priority

&#x20;   df\['TRAVEL\_MOTIVE\_1'] = req.travel\_motive\_1

&#x20;   df\['TRAVEL\_COMPANIONS\_NUM'] = req.travel\_companions\_num

&#x20;   

&#x20;   features = \[

&#x20;       'match\_score', 'p\_nature', 'p\_culture', 'p\_food', 'p\_shopping', 'p\_experience', 'p\_relaxation',

&#x20;       'place\_mean\_score', 'visit\_count', 'RESIDENCE\_TIME\_MIN\_mean', 'RCMDTN\_INTENTION\_mean', 'REVISIT\_YN\_mean', 'REVISIT\_INTENTION\_mean',

&#x20;       'nature', 'culture', 'food', 'shopping', 'experience', 'relaxation',

&#x20;       'AGE\_GRP', 'INCOME', 'TRAVEL\_MISSION\_PRIORITY', 'TRAVEL\_MOTIVE\_1', 'TRAVEL\_COMPANIONS\_NUM'

&#x20;   ]

&#x20;   

&#x20;   X = df\[features].copy()

&#x20;   for col in features:

&#x20;       if X\[col].isna().sum() > 0:

&#x20;           X\[col] = X\[col].fillna(X\[col].median())

&#x20;           

&#x20;   # 4) predicted\_score 추론 및 최종 점수(final\_score) 계산 (6:4 비중)

&#x20;   df\['predicted\_score'] = model.predict(X)

&#x20;   df\['final\_score'] = df\['match\_score'] \* 0.6 + (df\['predicted\_score'] / 5.0) \* 0.4

&#x20;   

&#x20;   # 5) Top-N 정렬

&#x20;   top\_df = df.sort\_values(by='final\_score', ascending=False).head(req.top\_n)

&#x20;   

&#x20;   # 6) 표준 JSON 응답 조립

&#x20;   recommendations = \[]

&#x20;   for rank, (\_, row) in enumerate(top\_df.iterrows(), start=1):

&#x20;       recommendations.append({

&#x20;           "rank": rank,

&#x20;           "place\_name": str(row\['장소명']),

&#x20;           "scores": {

&#x20;               "final\_score": round(float(row\['final\_score']), 3),

&#x20;               "match\_score": round(float(row\['match\_score']), 3),

&#x20;               "predicted\_score": round(float(row\['predicted\_score']), 2)

&#x20;           },

&#x20;           "place\_embeddings": {

&#x20;               "nature": float(row\['자연\_임베딩']),

&#x20;               "culture": float(row\['역사문화\_임베딩']),

&#x20;               "food": float(row\['미식\_임베딩']),

&#x20;               "shopping": float(row\['쇼핑\_임베딩']),

&#x20;               "experience": float(row\['체험\_임베딩']),

&#x20;               "relaxation": float(row\['휴식\_임베딩'])

&#x20;           },

&#x20;           "metadata": {

&#x20;               "mean\_score": round(float(row\['평균평점']), 1),

&#x20;               "visit\_count": int(row\['누적방문수'])

&#x20;           }

&#x20;       })

&#x20;       

&#x20;   return {

&#x20;       "user\_profile": u\_vec\_dict,

&#x20;       "total\_recommended": len(recommendations),

&#x20;       "recommendations": recommendations

&#x20;   }
```
